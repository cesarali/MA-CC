\section{Efficiency of Feedback Control}
\label{sec:feedback_efficiency}

We study one finite feedback cycle of the coarse population coordinate rather
than assigning an energetic thermodynamics to the complete language-model
system. Let $n_k\in\{0,\ldots,N\}$ be the number of agents supporting the
controller target $Z$ before cycle $k$, and let $x_k=n_k/N$. The causal order is
\[
n_k\longrightarrow Y_k\longrightarrow U_k\longrightarrow n_{k+1},
\]
where $Y_k$ is a finite sensor sample and $U_k\in\{0,1\}$ denotes NoOp or
advocacy. The theory separates three questions: how strongly advocacy moves the
population, how efficiently action-dependent information is converted into
coherent mean motion, and how efficiently the finite-time thermodynamic budget
is converted into directed current.

\subsection{Finite-information feedback controller}

The controller observes $q_c$ agents without replacement. If $Y_k$ is the
number of sampled agents supporting $Z$, then
\begin{equation}
S(y\mid n)
= P(Y_k=y\mid n_k=n)
=
\frac{\binom{n}{y}\binom{N-n}{q_c-y}}{\binom{N}{q_c}}.
\label{eq:sensor_kernel_single_affinity}
\end{equation}
The feedback policy is
\begin{equation}
\pi(1\mid y)
=\sigma\!\left[\beta\left(\theta-\frac{y}{q_c}\right)\right],
\qquad
\sigma(z)=\frac{1}{1+e^{-z}},
\label{eq:feedback_policy_single_affinity}
\end{equation}
so the advocacy probability conditional on the true population state is
\begin{equation}
a_n
\equiv P(U_k=1\mid n_k=n)
=\sum_y S(y\mid n)\pi(1\mid y).
\label{eq:advocacy_probability_single_affinity}
\end{equation}
Thus $q_c$, $\theta$, and $\beta$ determine when the controller acts.

\subsection{Single-affinity finite-compliance actuation}

A controlled microscopic opportunity is described by one directional affinity
$h$ and one kinetic compliance $\gamma\in[0,1]$. Writing $p_h=\sigma(h)$, the
count kernel is
\begin{align}
K_{h,\gamma}(n+1\mid n)
&=\gamma\frac{N-n}{N}p_h,\\
K_{h,\gamma}(n-1\mid n)
&=\gamma\frac{n}{N}(1-p_h),\\
K_{h,\gamma}(n\mid n)
&=1-K_{h,\gamma}(n+1\mid n)-K_{h,\gamma}(n-1\mid n).
\label{eq:single_affinity_kernel}
\end{align}
The affinity $h$ determines the preferred direction conditional on revision;
$\gamma$ determines how readily a controlled opportunity produces any revision.
No second opposing affinity is introduced.

\subsection{Control susceptibility and exact response}
\label{sec:control-susceptibility}

Conditional on the same initial state $x$, define the controller-induced
susceptibility
\begin{equation}
\boxed{
\chi(x)
\equiv
\mathbb{E}[x'\mid U=1,x]-\mathbb{E}[x'\mid U=0,x].}
\label{eq:control-susceptibility}
\end{equation}
For the finite-compliance $q=1$ controlled layer, $b$ controlled opportunities
give the exact response
\begin{equation}
\boxed{
\chi_{h,\gamma}(x)
=[\sigma(h)-x]\Lambda_{b,\gamma},
\qquad
\Lambda_{b,\gamma}
=1-\left(1-\frac{\gamma}{N}\right)^b.}
\label{eq:chi_finite_bias}
\end{equation}
Hence the zero-response set point is $x^\star=\sigma(h)$, while $\gamma$ only
rescales the response amplitude. Ordinary $q=1$ voter updates are conditionally
mean-neutral and therefore do not alter this advocacy-versus-NoOp mean response.

\subsection{Information--response efficiency}

For the isolated controlled layer define
\begin{equation}
Q_0(m\mid n)=\delta_{mn},
\qquad
Q_1(m\mid n)=K_{h,\gamma}^{\,b}(m\mid n),
\label{eq:controlled_layer_kernels}
\end{equation}
and
\begin{equation}
Q_\pi(m\mid n)
=(1-a_n)Q_0(m\mid n)+a_nQ_1(m\mid n).
\label{eq:policy_mixture_kernel}
\end{equation}
The exact state-local action-to-population information is
\begin{align}
T_\pi(n)
&\equiv I(U_k;n_{k+1}\mid n_k=n) \\
&=
\sum_{u\in\{0,1\}}P(u\mid n)
\sum_m Q_u(m\mid n)
\log_2\frac{Q_u(m\mid n)}{Q_\pi(m\mid n)}.
\label{eq:action-population-information}
\end{align}
Pinsker's inequality yields
\begin{equation}
T_\pi(n)
\ge
\frac{2a_n(1-a_n)}{\ln 2}\,
\chi_{h,\gamma}(n/N)^2,
\label{eq:information-response-bound}
\end{equation}
which motivates the bounded information--response efficiency
\begin{equation}
\boxed{
\eta_{\mathrm{IR}}(n)
=
\frac{2a_n(1-a_n)\chi_{h,\gamma}(n/N)^2}
{(\ln 2)T_\pi(n)}
\le 1.}
\label{eq:eta_ir}
\end{equation}
It measures how much of the action-dependent change in the full next-state
distribution is expressed as coherent mean target motion. The derivation is in
Appendix~\ref{app:information_response_bound}.

\paragraph{Finite-horizon aggregation.}
The efficiency above is defined locally at a fixed population state. In a
nonstationary experiment, the state occupancy changes from cycle to cycle. Let
\begin{equation}
B_{\mathrm{IR}}(n)
\equiv
\frac{2a_n(1-a_n)}{\ln 2}\,
\chi_{h,\gamma}(n/N)^2
\label{eq:ir_response_term}
\end{equation}
denote the state-local Pinsker response term. For a finite horizon of $H$
feedback cycles, we define the experiment-level information--response
efficiency by accumulating the state-weighted numerator and denominator before
forming the ratio:
\begin{equation}
\boxed{
\eta_{\mathrm{IR}}^{(H)}
=
\frac{
\sum_{k=0}^{H-1}\sum_n p_k(n)B_{\mathrm{IR}}(n)
}{
\sum_{k=0}^{H-1}\sum_n p_k(n)T_\pi(n)
}.}
\label{eq:eta_ir_finite_horizon}
\end{equation}
Because $B_{\mathrm{IR}}(n)\le T_\pi(n)$ state by state, the finite-horizon
ratio remains bounded, $0\le\eta_{\mathrm{IR}}^{(H)}\le1$. This quantity is
distinct from an arithmetic average of the state-local ratios
$\eta_{\mathrm{IR}}(n)$: it measures the fraction of the total
action-dependent information transmitted over the evaluation horizon that is
expressed through the response lower bound.

\subsection{Finite-time thermodynamic efficiency}
\label{sec:thermodynamic_efficiency}

The single affinity also gives a closed finite-time thermodynamic accounting.
Let $p_k(n)=P(n_k=n)$ and define the coarse system entropy
\begin{equation}
S_{\mathrm{sys}}[p]
=-\sum_n p(n)\ln p(n)
+\sum_n p(n)\ln\binom{N}{n}.
\label{eq:system_entropy_functional}
\end{equation}
The mean target-directed controlled current is
\begin{equation}
\boxed{
J_{c,k}
=N\Lambda_{b,\gamma}
\sum_n p_k(n)a_n
\left[\sigma(h)-\frac{n}{N}\right].}
\label{eq:mean_controlled_current_single_affinity}
\end{equation}
Let $I_{\mathrm{sens},k}\equiv I(n_k;Y_k)$ denote the information acquired by
the sensor. Comparing the forward feedback path measure with its finite-time
reverse reference gives the exact identity
\begin{equation}
\boxed{
\Delta S_{\mathrm{sys},k}
+hJ_{c,k}
+I_{\mathrm{sens},k}
=
\Sigma_k
=
D_{\mathrm{KL}}\!\left(P_F^{(k)}\Vert P_R^{(k)}\right)
\ge 0.}
\label{eq:finite_time_second_law}
\end{equation}
All terms in Eq.~(\ref{eq:finite_time_second_law}) use natural logarithms and
are therefore measured in nats. The transient boundary term
$\Delta S_{\mathrm{sys},k}$ is retained explicitly; no stationarity assumption
is required.

For target-directed operation, $hJ_{c,k}\ge0$. Removing the transient storage
term from the path irreversibility leaves the generalized finite-time control
expenditure
\begin{equation}
C_{\mathrm{th},k}
\equiv
\Sigma_k-\Delta S_{\mathrm{sys},k}
=
hJ_{c,k}+I_{\mathrm{sens},k}.
\label{eq:thermodynamic_control_expenditure}
\end{equation}
We therefore define the thermodynamic control efficiency as the fraction of
this non-storage expenditure expressed as affinity-weighted directed current:
\begin{equation}
\boxed{
\eta_{\mathrm{th},k}
\equiv
\frac{hJ_{c,k}}{\Sigma_k-\Delta S_{\mathrm{sys},k}}
=
\frac{hJ_{c,k}}{hJ_{c,k}+I_{\mathrm{sens},k}}
\le 1.}
\label{eq:eta_th}
\end{equation}
For $hJ_{c,k}\ge0$ and $C_{\mathrm{th},k}>0$,
$0\le\eta_{\mathrm{th},k}\le1$. At stationarity,
$\Delta S_{\mathrm{sys},k}=0$ and the same result reduces to
$\eta_{\mathrm{th}}=hJ_c/\Sigma$.

\paragraph{Finite-horizon aggregation.}
The quantities above are defined for one feedback cycle because the population
distribution $p_k(n)$ generally changes with $k$. For an experiment of $H$
feedback cycles, summing Eq.~(\ref{eq:finite_time_second_law}) gives
\begin{equation}
S_{\mathrm{sys}}[p_H]-S_{\mathrm{sys}}[p_0]
+h\sum_{k=0}^{H-1}J_{c,k}
+\sum_{k=0}^{H-1}I_{\mathrm{sens},k}
=\sum_{k=0}^{H-1}\Sigma_k,
\label{eq:finite_horizon_second_law}
\end{equation}
where the system-entropy increments telescope. The corresponding finite-horizon
non-storage expenditure is
\begin{equation}
C_{\mathrm{th}}^{(H)}
=h\sum_{k=0}^{H-1}J_{c,k}
+\sum_{k=0}^{H-1}I_{\mathrm{sens},k},
\label{eq:finite_horizon_expenditure}
\end{equation}
and the finite-horizon thermodynamic efficiency is
\begin{equation}
\boxed{
\eta_{\mathrm{th}}^{(H)}
=
\frac{h\sum_{k=0}^{H-1}J_{c,k}}
{h\sum_{k=0}^{H-1}J_{c,k}
+\sum_{k=0}^{H-1}I_{\mathrm{sens},k}}.}
\label{eq:eta_th_finite_horizon}
\end{equation}
Thus the experiment-level efficiency is a ratio of accumulated directed
current to accumulated non-storage expenditure, rather than the arithmetic
mean $H^{-1}\sum_k\eta_{\mathrm{th},k}$. This distinction is essential in the
nonstationary regime because the control expenditure can vary substantially
across cycles. Thus additional sensing lowers
thermodynamic efficiency whenever it increases acquired information without a
commensurate increase in directed current, whereas stronger effective actuation
raises the efficiency when it increases $hJ_c$ faster than the sensing term.
The transition-ratio and path-measure calculation leading to
Eqs.~(\ref{eq:finite_time_second_law})--(\ref{eq:eta_th}) is given in
Appendix~\ref{app:thermodynamic_efficiency_derivation}.
